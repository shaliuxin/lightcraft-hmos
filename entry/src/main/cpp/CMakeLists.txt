cmake_minimum_required(VERSION 3.5.0)
project(lightcraft_native)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(LIGHTCRAFT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
# OHOS_ARCH is the ABI hvigor is building (abiFilters in entry/build-profile.json5). This mapping
# is the single place in the C++ layer that knows about architectures.
if(OHOS_ARCH STREQUAL "arm64-v8a")
  set(RUST_TARGET aarch64-unknown-linux-ohos)
elseif(OHOS_ARCH STREQUAL "x86_64")
  set(RUST_TARGET x86_64-unknown-linux-ohos)
elseif(OHOS_ARCH STREQUAL "armeabi-v7a")
  set(RUST_TARGET armv7-unknown-linux-ohos)
else()
  message(FATAL_ERROR "LightCraft does not support OHOS_ARCH=${OHOS_ARCH}")
endif()
if(NOT DEFINED LIGHTCRAFT_RUST_PROFILE)
  set(LIGHTCRAFT_RUST_PROFILE release)
endif()
set(RUST_LIBRARY "${LIGHTCRAFT_ROOT}/native/rust/target/${RUST_TARGET}/${LIGHTCRAFT_RUST_PROFILE}/liblightcraft_hmos.a")
if(NOT EXISTS "${RUST_LIBRARY}")
  message(FATAL_ERROR "Run scripts/build_native.sh before building HAP: ${RUST_LIBRARY}")
endif()
add_library(lightcraft SHARED bridge.cpp)
target_link_libraries(lightcraft PRIVATE "${RUST_LIBRARY}" ace_napi.z ace_ndk.z native_window native_vsync hilog_ndk.z EGL GLESv3 dl m pthread)
target_link_options(lightcraft PRIVATE "-Wl,--gc-sections" "-Wl,--exclude-libs,ALL")
